cmake_minimum_required(VERSION 3.22)
project(retuned_randomizer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RANDOMIZER_SOURCE "" CACHE PATH "Pinned source from fetch_source.py")
if(NOT EXISTS "${RANDOMIZER_SOURCE}/source-receipt.json")
  message(FATAL_ERROR "Fetch the pinned source with tools/randomizer/fetch_source.py first")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py" "${RANDOMIZER_SOURCE}" COMMAND_ERROR_IS_FATAL ANY)
if(TARGET nlohmann_json::nlohmann_json)
  get_target_property(JSON_INCLUDE_DIR nlohmann_json::nlohmann_json INTERFACE_INCLUDE_DIRECTORIES)
else()
  find_path(JSON_INCLUDE_DIR nlohmann/json.hpp HINTS "${nlohmann_json_SOURCE_DIR}/include" REQUIRED)
endif()
set(UPSTREAM "${RANDOMIZER_SOURCE}/externals/oot3d-randomizer")
file(GLOB_RECURSE GENERATOR_SOURCES CONFIGURE_DEPENDS "${UPSTREAM}/source/*.cpp")
list(FILTER GENERATOR_SOURCES EXCLUDE REGEX "/(main|menu|music|patch|preset|utils|globals)\\.cpp$")
list(FILTER GENERATOR_SOURCES EXCLUDE REGEX "/custom_music/")
add_library(retuned_randomizer STATIC ${GENERATOR_SOURCES} "${UPSTREAM}/shared/shared_data.c" generator.cpp platform.cpp)
target_compile_definitions(retuned_randomizer PRIVATE OOTR_HOST)
target_include_directories(retuned_randomizer PRIVATE "${RANDOMIZER_SOURCE}/tools/oot3d/randomizer/host" "${UPSTREAM}/source" "${UPSTREAM}/shared" "${JSON_INCLUDE_DIR}")
target_include_directories(retuned_randomizer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(retuned-seed main.cpp)
target_link_libraries(retuned-seed PRIVATE retuned_randomizer)
include(CTest)
if(BUILD_TESTING)
  add_executable(document-transfer-test document_transfer_test.cpp)
  if(MSVC)
    target_compile_options(document-transfer-test PRIVATE /UNDEBUG)
  else()
    target_compile_options(document-transfer-test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME randomizer-document-transfer COMMAND document-transfer-test)
  add_executable(android-session-test android_session_test.cpp)
  add_executable(legacy-hud-anchor-test legacy_hud_anchor_test.cpp)
  foreach(test android-session-test legacy-hud-anchor-test)
    if(MSVC)
      target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
  endforeach()
  add_test(NAME randomizer-android-session COMMAND android-session-test)
  add_test(NAME randomizer-legacy-hud-anchor COMMAND legacy-hud-anchor-test)
  add_executable(continue-game-test continue_game_test.cpp)
  if(MSVC)
    target_compile_options(continue-game-test PRIVATE /UNDEBUG)
  else()
    target_compile_options(continue-game-test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME randomizer-native-continue COMMAND continue-game-test)
  add_executable(generator-test generator_test.cpp)
  target_link_libraries(generator-test PRIVATE retuned_randomizer)
  target_include_directories(generator-test PRIVATE "${JSON_INCLUDE_DIR}")
  add_test(NAME generator-isolation COMMAND generator-test)
endif()

set(RANDOMIZER_ENGINE "" CACHE PATH "Existing native engine for isolated runtime-service tests")
if(RANDOMIZER_ENGINE)
  set(MODULE_ROOT "${RANDOMIZER_ENGINE}/runtime/triaevum_module")
  if(NOT EXISTS "${MODULE_ROOT}/src/sha256.cpp")
    message(FATAL_ERROR "RANDOMIZER_ENGINE must contain the native module SHA-256 implementation")
  endif()
  add_library(retuned_randomizer_services STATIC save_store.cpp runtime_services.cpp run_catalog.cpp run_launcher.cpp seed_activation.cpp run_transfer.cpp
    "${MODULE_ROOT}/src/sha256.cpp")
  target_include_directories(retuned_randomizer_services PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    PRIVATE "${MODULE_ROOT}/include" "${JSON_INCLUDE_DIR}")
  add_library(retuned_randomizer_frontend INTERFACE)
  target_link_libraries(retuned_randomizer_frontend INTERFACE retuned_randomizer retuned_randomizer_services)
  target_include_directories(retuned_randomizer_frontend INTERFACE "${JSON_INCLUDE_DIR}")
  if(BUILD_TESTING)
    add_executable(run-transfer-test run_transfer_test.cpp)
    target_link_libraries(run-transfer-test PRIVATE retuned_randomizer_frontend)
    add_test(NAME randomizer-run-transfer COMMAND run-transfer-test)
    add_executable(seed-activation-test seed_activation_test.cpp)
    target_include_directories(seed-activation-test PRIVATE "${JSON_INCLUDE_DIR}")
    target_link_libraries(seed-activation-test PRIVATE retuned_randomizer_services retuned_randomizer)
    add_test(NAME randomizer-seed-activation COMMAND seed-activation-test)
    add_executable(runtime-services-test runtime_services_test.cpp)
    target_link_libraries(runtime-services-test PRIVATE retuned_randomizer_services)
    add_test(NAME randomizer-save-services COMMAND runtime-services-test)
    add_executable(run-catalog-test run_catalog_test.cpp)
    target_include_directories(run-catalog-test PRIVATE "${JSON_INCLUDE_DIR}")
    target_link_libraries(run-catalog-test PRIVATE retuned_randomizer_services retuned_randomizer)
    add_test(NAME randomizer-run-catalog COMMAND run-catalog-test)
    add_executable(run-launcher-test run_launcher_test.cpp)
    target_include_directories(run-launcher-test PRIVATE "${JSON_INCLUDE_DIR}" "${MODULE_ROOT}/include")
    target_link_libraries(run-launcher-test PRIVATE retuned_randomizer_services retuned_randomizer)
    add_test(NAME randomizer-run-launcher COMMAND run-launcher-test)
  endif()
endif()
